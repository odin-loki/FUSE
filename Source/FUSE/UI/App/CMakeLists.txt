# fuse_app_native: real OS windows (X11 via runtime-loaded libX11, Win32, Android NativeActivity) + Vulkan swapchain
# presenter, custom window chrome, shell services (tray icon, notifications), platform services (DPI, displays,
# cursors, theme, dialogs).
if(ANDROID)
    set(_glue "${ANDROID_NDK}/sources/android/native_app_glue")
    add_library(fuse_app_native STATIC src/native_android.cpp src/platform_posix.cpp "${_glue}/android_native_app_glue.c")
    target_include_directories(fuse_app_native PRIVATE "${_glue}")
    target_link_libraries(fuse_app_native PUBLIC fuse_app android log)
    # the glue's ANativeActivity_onCreate lives in this static library: keep it in the app's shared library
    target_link_options(fuse_app_native INTERFACE "-u" "ANativeActivity_onCreate")
    # fuse::platform lifecycle / surface-loss callbacks: fuse_core in tree, its platform source standalone
    if(TARGET fuse_core)
        target_link_libraries(fuse_app_native PUBLIC fuse_core)
    else()
        set(_core "${FUSE_UI_REPO_ROOT}/Source/FUSE/Core")
        add_library(fuse_app_platform STATIC "${_core}/src/platform/platform_hardening.cpp")
        target_include_directories(fuse_app_platform PUBLIC "${_core}/include")
        target_link_libraries(fuse_app_native PUBLIC fuse_app_platform)
    endif()
    target_compile_definitions(fuse_app_native PRIVATE FUSE_APP_HAS_PLATFORM_LIFECYCLE=1)
else()
    add_library(fuse_app_native STATIC src/native.cpp src/platform_win32.cpp src/platform_posix.cpp)
    target_link_libraries(fuse_app_native PUBLIC fuse_app)
endif()
# platform services feed fuse::platform v2 events (header-only events_v2 / platform_bridge) into the UI
target_include_directories(fuse_app_native PRIVATE ${FUSE_UI_REPO_ROOT}/Source/FUSE/Core/include)
if(FUSE_UI_VK_LINK)
    target_sources(fuse_app_native PRIVATE src/vk_present.cpp)
    target_link_libraries(fuse_app_native PRIVATE ${FUSE_UI_VK_LINK})
    target_compile_definitions(fuse_app_native PRIVATE FUSE_APP_HAS_VULKAN=1)
    if(TARGET fuse_ui_vk)
        target_link_libraries(fuse_app_native PRIVATE fuse_ui_vk)
        target_compile_definitions(fuse_app_native PRIVATE FUSE_APP_HAS_UI_VK=1)
    endif()
endif()
if(WIN32 OR MINGW)
    target_sources(fuse_app_native PRIVATE src/chrome_win32.cpp src/shell_win32.cpp)
    target_link_libraries(fuse_app_native PRIVATE user32 gdi32 dwmapi ole32 shell32 advapi32 uuid)
else()
    target_sources(fuse_app_native PRIVATE src/shell_unix.cpp) # Android: no tray, notifications unavailable
    target_link_libraries(fuse_app_native PRIVATE ${CMAKE_DL_LIBS})
endif()

option(FUSE_BUILD_APP_SAMPLES "Build the fuse_app samples (hello_app, notes_app, fuse_ui_android_demo; game_menu comes with fuse_ui_game)" ON)
if(FUSE_BUILD_APP_SAMPLES)
    if(ANDROID)
        # APKs: <target>.apk in the build directory (fuse_app_add_android_app)
        fuse_app_add_android_app(hello_app SOURCES samples/hello_app.cpp PACKAGE org.fuse.hello LABEL "FUSE Hello"
                                 ASSETS "ui=${CMAKE_CURRENT_SOURCE_DIR}/samples/ui")
        target_compile_definitions(hello_app PRIVATE FUSE_APP_SAMPLE_UI_DIR="ui")
        fuse_app_add_android_gate(fuse_app_hello_startup_android hello_app org.fuse.hello startup)
        fuse_app_add_android_app(fuse_ui_android_demo SOURCES samples/android_demo.cpp PACKAGE org.fuse.uidemo
                                 LABEL "FUSE UI")
        fuse_app_add_android_gates(fuse_ui_android_demo org.fuse.uidemo)
    else()
        foreach(s hello_app notes_app)
            add_executable(${s} samples/${s}.cpp)
            target_link_libraries(${s} PRIVATE fuse_app_native)
            target_compile_definitions(${s} PRIVATE FUSE_APP_SAMPLE_UI_DIR="${CMAKE_CURRENT_SOURCE_DIR}/samples/ui")
            if(TARGET fuse_ui_icons) # the starter icon pack (toolbar / menu / button icons)
                add_dependencies(${s} fuse_ui_icons)
                target_compile_definitions(${s} PRIVATE FUSE_APP_SAMPLE_ICONS="${FUSE_UI_ICON_PACK}")
            endif()
        endforeach()
        add_executable(fuse_ui_android_demo samples/android_demo.cpp)
        target_link_libraries(fuse_ui_android_demo PRIVATE fuse_app_native)
    endif()
endif()

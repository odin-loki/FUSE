# fuse_app (plan UI5): application shell for UI-only projects + samples.
add_library(fuse_app STATIC src/app.cpp)
target_include_directories(fuse_app PUBLIC include PRIVATE ${FUSE_UI_REPO_ROOT}/vendor/stb)
target_link_libraries(fuse_app PUBLIC fuse_ui_doc fuse_ui_os)

# fuse_app_native: real OS windows (X11 via runtime-loaded libX11, Win32) + Vulkan swapchain presenter.
add_library(fuse_app_native STATIC src/native.cpp src/wayland.cpp)
if(ANDROID) # G4: NativeActivity glue from the NDK
    enable_language(C)
    add_library(fuse_android_glue STATIC ${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c)
    target_include_directories(fuse_android_glue PUBLIC ${ANDROID_NDK}/sources/android/native_app_glue)
    target_sources(fuse_app_native PRIVATE src/android.cpp)
    target_link_libraries(fuse_app_native PUBLIC fuse_android_glue android log)
endif()
if(IOS) # UIKit backend: UNVERIFIED (never built; written on Linux)
    enable_language(OBJCXX)
    target_sources(fuse_app_native PRIVATE src/uikit.mm)
    set_source_files_properties(src/uikit.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
    target_link_libraries(fuse_app_native PRIVATE "-framework UIKit" "-framework QuartzCore")
elseif(APPLE) # G3 Cocoa backend: UNVERIFIED (never built; written on Linux)
    enable_language(OBJCXX)
    target_sources(fuse_app_native PRIVATE src/cocoa.mm)
    set_source_files_properties(src/cocoa.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
    target_link_libraries(fuse_app_native PRIVATE "-framework Cocoa" "-framework QuartzCore")
endif()
target_link_libraries(fuse_app_native PUBLIC fuse_app)
if(FUSE_UI_VK_LINK)
    target_sources(fuse_app_native PRIVATE src/vk_present.cpp)
    target_link_libraries(fuse_app_native PRIVATE ${FUSE_UI_VK_LINK})
    target_compile_definitions(fuse_app_native PRIVATE FUSE_APP_HAS_VULKAN=1)
    if(TARGET fuse_ui_vk)
        target_link_libraries(fuse_app_native PRIVATE fuse_ui_vk)
        target_compile_definitions(fuse_app_native PRIVATE FUSE_APP_HAS_UI_VK=1)
    endif()
endif()
if(WIN32 OR MINGW)
    target_sources(fuse_app_native PRIVATE src/win32_dnd.cpp)
    target_link_libraries(fuse_app_native PRIVATE user32 gdi32 ole32 shell32 uuid)
else()
    target_link_libraries(fuse_app_native PRIVATE ${CMAKE_DL_LIBS})
endif()

option(FUSE_BUILD_APP_SAMPLES "Build the fuse_app samples (hello_app, notes_app)" ON)
if(FUSE_BUILD_APP_SAMPLES)
    foreach(s hello_app notes_app)
        add_executable(${s} samples/${s}.cpp)
        target_link_libraries(${s} PRIVATE fuse_app_native)
        target_compile_definitions(${s} PRIVATE FUSE_APP_SAMPLE_UI_DIR="${CMAKE_CURRENT_SOURCE_DIR}/samples/ui")
    endforeach()
endif()


if(ANDROID AND FUSE_BUILD_APP_SAMPLES)
    # libfuse_android_hello.so for android/build_apk.sh (NativeActivity: android_main must stay exported)
    add_library(fuse_android_hello SHARED samples/android_hello.cpp)
    target_link_libraries(fuse_android_hello PRIVATE fuse_app_native)
    target_link_options(fuse_android_hello PRIVATE "-u" "ANativeActivity_onCreate")
endif()

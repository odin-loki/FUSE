**Duration:** 6–7 weeks with AI-assisted coding

## 5.1 — Rendering Architecture Upgrade

Phase 2 built a thin hybrid rasteriser. Phase 5 replaces the placeholder shading with a production-grade deferred pipeline and expands the CUDA ray marcher into a full lighting and GI system.

### Full Frame Pipeline

```
Frame N:

[Vulkan — Graphics Queue]
├── PASS 1: Depth Prepass
│   └── Render all opaque geometry to depth buffer only (no shading)
│       Early-Z rejection eliminates overdraw in G-buffer pass
│
├── PASS 2: G-Buffer Pass
│   └── Render opaque geometry → normal, albedo, roughness, metallic, velocity, depth
│
├── PASS 3: Shadow Maps (cascaded, for triangle geometry)
│   └── 4 cascades rendered for directional light
│
└── Signal(vk_to_cuda, frame_N)

[CUDA — Compute Stream: Render]
├── SDF Ray March    — exact surface detail, fills gaps in raster
├── DDGI Probe Update — irradiance probes via ray casting
├── Clustered Light  — assign lights to screen-space clusters
├── Deferred Shading — PBR shading per pixel using G-buffer + probes + clusters
├── SDF Shadows      — exact soft shadows for SDF geometry
├── Screen-Space AO  — HBAO from depth buffer
└── Signal(cuda_to_vk, frame_N)

[Vulkan — Graphics Queue]
├── PASS 4: Transparent Pass   — forward+ shading for alpha geometry
├── PASS 5: Atmosphere/Sky
├── PASS 6: TAA Resolve        — temporal accumulation
├── PASS 7: Post-Process Stack — bloom, lens flare, DoF, motion blur, tonemapping
└── PASS 8: UI Composite + Present
```

## 5.2 — G-Buffer Layout (Revised)

Expanded from Phase 2 to carry all data needed for deferred PBR shading.

```
RT0: RGBA16F   — World-space normal (xyz, oct-encoded) | AO (w)
RT1: RGBA8     — Albedo/base colour (rgb) | Alpha (a)
RT2: RGBA8     — Roughness (r) | Metallic (g) | Emissive mask (b) | Shading model ID (a)
RT3: RG16F     — Screen-space velocity (motion vectors for TAA)
RT4: R32F      — Depth (reversed-Z, 32-bit float)
RT5: RGBA16F   — Emissive colour (rgb) | unused (w)  — only written by emissive surfaces
```

### G-Buffer Encoding Helpers

```glsl
// shaders/common/gbuffer.glsl

// Octahedral normal encoding — 2 floats instead of 3, fully reversible
vec2 encode_normal(vec3 n) {
  n /= (abs(n.x) + abs(n.y) + abs(n.z));
  vec2 o = n.z >= 0.0 ? n.xy : (1.0 - abs(n.yx)) * sign(n.xy);
  return o * 0.5 + 0.5;
}

vec3 decode_normal(vec2 enc) {
  enc = enc * 2.0 - 1.0;
  vec3 n = vec3(enc.xy, 1.0 - abs(enc.x) - abs(enc.y));
  if (n.z < 0.0) n.xy = (1.0 - abs(n.yx)) * sign(n.xy);
  return normalize(n);
}

// Pack G-buffer from fragment shader outputs
void write_gbuffer(
  vec3 normal, vec3 albedo, f32 roughness, f32 metallic,
  vec3 emissive, float ao, vec2 velocity, uint shading_model
) {
  out_rt0 = vec4(encode_normal(normal), ao, 0.0);
  out_rt1 = vec4(albedo, 1.0);
  out_rt2 = vec4(roughness, metallic, length(emissive) > 0.001 ? 1.0 : 0.0,
                 float(shading_model) / 255.0);
  out_rt3 = vec4(velocity, 0.0, 0.0);
  out_rt5 = vec4(emissive, 0.0);
}
```

## 5.3 — PBR Material System

### BRDF — GGX Microfacet Model

The engine uses the same BRDF model as Unreal Engine 4 — GGX specular with Smith masking, Lambert diffuse — but implemented analytically with no texture-baked approximations.

```glsl
// shaders/common/brdf.glsl

const float PI = 3.14159265358979;

// GGX normal distribution function — D term
float D_GGX(float NoH, float roughness) {
  float a  = roughness * roughness;
  float a2 = a * a;
  float d  = (NoH * a2 - NoH) * NoH + 1.0;
  return a2 / (PI * d * d);
}

// Smith masking-shadowing — G term
float G_SmithGGX(float NoV, float NoL, float roughness) {
  float a  = roughness * roughness;
  float gv = NoL * sqrt(NoV * NoV * (1.0 - a) + a);
  float gl = NoV * sqrt(NoL * NoL * (1.0 - a) + a);
  return 0.5 / max(gv + gl, 1e-5);
}

// Schlick Fresnel — F term
vec3 F_Schlick(float VoH, vec3 F0) {
  return F0 + (1.0 - F0) * pow(1.0 - VoH, 5.0);
}

// Full Cook-Torrance BRDF evaluation
vec3 brdf_evaluate(
  vec3  albedo,
  float roughness,
  float metallic,
  vec3  N,           // surface normal
  vec3  V,           // view direction
  vec3  L,           // light direction
  vec3  light_color,
  float light_intensity
) {
  vec3  H    = normalize(V + L);
  float NoV  = max(dot(N, V), 1e-4);
  float NoL  = max(dot(N, L), 0.0);
  float NoH  = max(dot(N, H), 0.0);
  float VoH  = max(dot(V, H), 0.0);

  vec3  F0   = mix(vec3(0.04), albedo, metallic);
  vec3  F    = F_Schlick(VoH, F0);
  float D    = D_GGX(NoH, roughness);
  float G    = G_SmithGGX(NoV, NoL, roughness);

  vec3  spec = (D * G * F) / max(4.0 * NoV * NoL, 1e-5);
  vec3  diff = (1.0 - F) * (1.0 - metallic) * albedo / PI;

  return (diff + spec) * light_color * light_intensity * NoL;
}
```

## 5.4 — Clustered Deferred Shading

O(n) lighting complexity regardless of light count. Screen space divided into a 3D cluster grid. Lights are assigned to clusters. Shading reads only the lights touching the cluster of each pixel.

### Deferred Shade Kernel Body

```cpp
__global__ void deferred_shade_kernel(/* params */) {
  u32 px = blockIdx.x * blockDim.x + threadIdx.x;
  u32 py = blockIdx.y * blockDim.y + threadIdx.y;
  if (px >= width || py >= height) return;

  // Reconstruct world position from depth
  float depth;
  surf2Dread(&depth, gbuf_depth, px * sizeof(float), py);
  if (depth <= 0.f) { surf2Dwrite(make_float4(0,0,0,1), output, px*16, py); return; }

  vec2 ndc = {2.f*px/width - 1.f, 1.f - 2.f*py/height};
  vec4 clip = {ndc.x, ndc.y, depth, 1.f};
  vec4 world = inv_view_proj * clip;
  vec3 world_pos = {world.x/world.w, world.y/world.w, world.z/world.w};
  vec3 V = (cam_pos - world_pos).normalized();

  // Read G-buffer
  float4 n_ao  = surf2Dread<float4>(gbuf_normal_ao,          px*16, py);
  float4 alb   = surf2Dread<float4>(gbuf_albedo,             px*16, py);
  float4 rm    = surf2Dread<float4>(gbuf_roughness_metallic, px*16, py);
  float4 emit  = surf2Dread<float4>(gbuf_emissive,           px*16, py);

  vec3 N        = decode_normal_cuda({n_ao.x, n_ao.y});
  f32  ao       = n_ao.z;
  vec3 albedo   = {alb.x, alb.y, alb.z};
  f32  roughness = rm.x;
  f32  metallic  = rm.y;

  vec3 Lo = {0,0,0};

  // Directional light (sun) — analytic
  Lo = Lo + brdf_evaluate_cuda(albedo, roughness, metallic, N, V,
                                sun_direction.normalized(), sun_color, sun_intensity);

  // Clustered point lights
  // Determine cluster index from screen position and depth
  u32 cluster_z   = (u32)(logf(depth / near_plane) / logf(far_plane / near_plane) * slices_z);
  u32 cluster_idx = (py/height * tiles_y) * tiles_x * slices_z
                  + (px/width  * tiles_x) * slices_z + cluster_z;
  cluster_idx = min(cluster_idx, tiles_x * tiles_y * slices_z - 1);

  uint2 grid_entry = light_grid[cluster_idx];
  u32   light_offset = grid_entry.x;
  u32   light_count_local = grid_entry.y;

  for (u32 li = 0; li < light_count_local; li++) {
    u32         lidx  = light_list[light_offset + li];
    GPUPointLight& pl = point_lights[lidx];
    vec3  L    = (pl.position - world_pos).normalized();
    f32   dist = (pl.position - world_pos).length();
    f32   atten= 1.f / (1.f + dist*dist / (pl.radius*pl.radius));
    Lo = Lo + brdf_evaluate_cuda(albedo, roughness, metallic, N, V, L,
                                  pl.color, pl.intensity * atten);
  }

  // GI — sample nearest irradiance probes (DDGI)
  vec3 irradiance = sample_irradiance_probes(world_pos, N,
                                              irradiance_probes,
                                              probe_grid_origin,
                                              probe_grid_spacing,
                                              probe_grid_dims);
  vec3 diffuse_gi = albedo * irradiance * ao / PI;
  Lo = Lo + diffuse_gi;

  // Emissive
  Lo = Lo + vec3{emit.x, emit.y, emit.z};

  surf2Dwrite(make_float4(Lo.x, Lo.y, Lo.z, 1.f), output, px*16, py);
}
```

### SDF Soft Shadows (CUDA)

Exact soft shadows for all SDF geometry — analytically correct penumbra, no filtering hacks.

```cpp
// compute/kernels/sdf_shadows.cuh

// Soft shadow via penumbra ray marching — analytically correct
// Returns shadow factor in [0,1], 1 = fully lit, 0 = fully shadowed
__device__ f32 sdf_soft_shadow(
  auto        scene_sdf,   // callable — same SDF as ray marcher
  vec3        ray_origin,
  vec3        ray_dir,
  f32         t_min,
  f32         t_max,
  f32         penumbra_k   // higher k = sharper shadow
) {
  f32 res  = 1.f;
  f32 t    = t_min;
  f32 prev = 1e30f;

  for (u32 i = 0; i < 64 && t < t_max; i++) {
    vec3 p = {ray_origin.x + ray_dir.x*t,
              ray_origin.y + ray_dir.y*t,
              ray_origin.z + ray_dir.z*t};
    f32 d = scene_sdf(p);

    if (d < 0.0001f) return 0.f;  // occluded

    // Improved soft shadow — accounts for previous step (Iñigo Quilez)
    f32 y = d*d / (2.f * prev);
    f32 dd = sqrtf(d*d - y*y);
    res  = fminf(res, penumbra_k * dd / fmaxf(0.f, t - y));
    prev = d;
    t   += d;
  }
  return clamp(res, 0.f, 1.f);
}

// Shadow kernel — one thread per pixel, traces shadow ray to sun
__global__ void sdf_shadow_kernel(
  cudaSurfaceObject_t  depth_buffer,
  cudaSurfaceObject_t  normal_buffer,
  cudaSurfaceObject_t  out_shadow,
  mat4                 inv_view_proj,
  vec3                 sun_direction,
  const RayMarchParams::SDFObject* objects,
  u32                  object_count,
  f32                  penumbra_k,
  u32                  width, u32 height
);
```

## 5.6 — Global Illumination: DDGI

Dynamic Diffuse Global Illumination via irradiance probes. Probes are placed on a 3D grid. Each frame a subset of probes is updated by casting rays from the probe position and accumulating radiance. Probes store spherical harmonic-encoded irradiance — correct low-frequency diffuse GI with zero flickering.

### Probe Grid

```cpp
  f32  atmo_radius        = 6471000.f;
  vec3 rayleigh_coeff     = {5.8e-6f, 13.5e-6f, 33.1e-6f};  // RGB
  f32  mie_coeff          = 21e-6f;
  f32  rayleigh_scale_h   = 8000.f;
  f32  mie_scale_h        = 1200.f;
  f32  mie_scatter_dir    = 0.758f;      // Henyey-Greenstein g parameter
  u32  view_samples       = 16;
  u32  light_samples      = 8;
};

// Rayleigh phase function — exact
__device__ f32 rayleigh_phase(f32 cos_theta) {
  return (3.f / (16.f * 3.14159f)) * (1.f + cos_theta * cos_theta);
}

// Mie phase function — Henyey-Greenstein approximation
__device__ f32 mie_phase(f32 cos_theta, f32 g) {
  f32 g2 = g * g;
  return (3.f / (8.f * 3.14159f)) * ((1.f - g2) * (1.f + cos_theta*cos_theta))
       / ((2.f + g2) * powf(1.f + g2 - 2.f*g*cos_theta, 1.5f));
}

// Single-scattering sky integral along view ray
__device__ vec3 compute_sky_colour(
  vec3 ray_origin, vec3 ray_dir, vec3 sun_direction,
  const AtmosphereParams& p
);
```

## 5.9 — Temporal Anti-Aliasing

TAA accumulates multiple sub-pixel samples over time, producing noise-free high-quality output with history rejection for fast-moving objects.

```cpp
// renderer/taa/taa.hpp
#pragma once
#include "../resources.hpp"

struct TAAParams {
  f32  blend_factor     = 0.1f;    // current frame weight — lower = more stable
  f32  velocity_rejection= 0.9f;   // velocity-weighted history rejection
  f32  depth_rejection  = 0.1f;    // depth-based disocclusion rejection
  f32  clamp_gamma      = 1.25f;   // variance clamp extent
  bool use_catmull_rom  = true;    // higher quality history sampling
};

class TAA {
public:
  void init(u32 width, u32 height, ResourceManager& resources);
  void destroy();
  void resolve(
    cudaSurfaceObject_t current_frame,
    cudaSurfaceObject_t velocity_buffer,
    cudaSurfaceObject_t depth_buffer,
    cudaSurfaceObject_t output,
    const TAAParams& params,
    cudaStream_t stream
  );

  // Jitter offsets — returned as vec2 NDC offset to apply to projection matrix
  vec2 current_jitter() const;
  void advance_jitter();

private:
  TextureHandle history_buffer_[2];
  u32           history_idx_     = 0;
  u32           jitter_idx_      = 0;
  // Halton sequence — 8-frame repeating
  static constexpr f32 HALTON_X[8] = {0.5f, 0.25f, 0.75f, 0.125f, 0.625f, 0.375f, 0.875f, 0.0625f};
  static constexpr f32 HALTON_Y[8] = {0.333f, 0.667f, 0.111f, 0.444f, 0.778f, 0.222f, 0.556f, 0.889f};
};
```

### TAA Resolve Kernel

```cpp
// renderer/taa/taa_kernels.cuh

__global__ void taa_resolve_kernel(
  cudaSurfaceObject_t current,
  cudaSurfaceObject_t history,
  cudaSurfaceObject_t velocity,
  cudaSurfaceObject_t depth,
  cudaSurfaceObject_t output,
  TAAParams           params,
  u32                 width, u32 height
) {
  u32 px = blockIdx.x * blockDim.x + threadIdx.x;
  u32 py = blockIdx.y * blockDim.y + threadIdx.y;
  if (px >= width || py >= height) return;

  float4 cur  = surf2Dread<float4>(current, px*16, py);
  float2 vel  = surf2Dread<float2>(velocity, px*8,  py);

  // Reproject history sample via motion vector
  f32 hist_px = (f32)px - vel.x * width;
  f32 hist_py = (f32)py - vel.y * height;

  // Catmull-Rom bicubic history sample
  float4 hist = catmull_rom_sample(history, hist_px, hist_py, width, height);

  // Variance clamp — prevent ghosting
  // Compute neighbourhood min/max in a 3×3 window
  float4 n_min = cur, n_max = cur;
  for (i32 dy = -1; dy <= 1; dy++) for (i32 dx = -1; dx <= 1; dx++) {
    float4 s = surf2Dread<float4>(current,
                                   clamp(px+dx,0u,width-1)*16,
                                   clamp(py+dy,0u,height-1));
    n_min = fminf4(n_min, s);
    n_max = fmaxf4(n_max, s);
  }
  // Expand by gamma
  float4 centre = (n_min + n_max) * 0.5f;
  float4 extent = (n_max - n_min) * 0.5f * params.clamp_gamma;
  hist = clampf4(hist, centre - extent, centre + extent);

  // Velocity rejection
  f32 vel_len = sqrtf(vel.x*vel.x + vel.y*vel.y);
  f32 blend   = params.blend_factor + vel_len * params.velocity_rejection;
  blend       = clampf(blend, params.blend_factor, 1.f);

  float4 result = lerp4(hist, cur, blend);
  surf2Dwrite(result, output, px*16, py);
}
```

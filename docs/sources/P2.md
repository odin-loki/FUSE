### Frame-in-Flight Management

Triple-buffered. Frame N+1 never waits for frame N-1 to finish presenting. Each frame has its own command pool, descriptor pool, and per-frame scratch allocator.

## 2.3 — Resource System & Bindless Architecture

Bindless from day one. No per-draw descriptor set updates. All textures, buffers, and samplers live in a global descriptor table indexed by u32. This is the architecture modern engines (Unreal 5, id Tech 7) moved to and it maps perfectly to CUDA's flat memory model.

## 2.5 — Command Buffer & Render Graph

A lightweight render graph drives the frame. Nodes declare their resource reads and writes. The graph resolves image layout transitions, pipeline barriers, and pass ordering automatically. No manual `vkCmdPipelineBarrier` calls outside the graph.

### Render Graph

```cpp
// renderer/render_graph.hpp
#pragma once
#include "resources.hpp"
#include "../core/types.hpp"
#include <functional>
#include <span>

enum class RGResourceAccess : u32 {
  ColorAttachmentWrite,
  DepthAttachmentWrite,
  ShaderRead,
  ShaderWrite,      // storage image / SSBO
  TransferSrc,
  TransferDst,
  Present,
  CUDAWrite,        // resource will be written by a CUDA kernel this frame
  CUDARead,
};

struct RGTextureRef { u32 id; };
struct RGBufferRef  { u32 id; };

struct RGPassDesc {
  const char* name;
  std::function<void(VkCommandBuffer)> execute;
  // Declared resource accesses — graph resolves barriers
  std::span<std::pair<RGTextureRef, RGResourceAccess>> texture_accesses;
  std::span<std::pair<RGBufferRef,  RGResourceAccess>> buffer_accesses;
  bool is_compute  = false;
  bool is_cuda     = false;   // pass executes on CUDA, not Vulkan
};

class RenderGraph {
public:
  void begin_frame(u32 backbuffer_index);
  void end_frame(VkCommandBuffer cmd);

  RGTextureRef import_texture(TextureHandle h, VkImageLayout current_layout);
  RGBufferRef  import_buffer(BufferHandle h);
  RGTextureRef create_transient(TextureDesc desc);   // lives only this frame

  void add_pass(RGPassDesc desc);

  // Compiles graph: sorts passes, inserts barriers, culls unused passes
  void compile();
  // Executes all passes in order, submitting to appropriate queues
  void execute(const VulkanDevice& device, FrameData& frame);

private:
  struct PassNode {
    RGPassDesc desc;
    u32        order;        // topological sort index
    bool       culled;
  };
  std::vector<PassNode> passes_;
  // Resource state tracking for barrier generation
  std::unordered_map<u32, VkImageLayout>       texture_layouts_;
  std::unordered_map<u32, VkAccessFlags2>      buffer_access_;
};

### Timeline Semaphore Sync

```cpp
// compute/cuda_vk_sync.hpp
#pragma once
#include "../core/types.hpp"
#include <vulkan/vulkan.h>
#include <cuda_runtime.h>

// Wrapper around a shared timeline semaphore
// Vulkan signals it after render pass, CUDA waits before compute
// CUDA signals it after compute, Vulkan waits before next pass
struct SharedTimeline {
  VkSemaphore           vk_semaphore;
  cudaExternalSemaphore_t cuda_semaphore;

  static SharedTimeline create(VkDevice vk_device);
  void                  destroy(VkDevice vk_device);

  void vk_signal  (VkQueue queue, u64 signal_value);
  void vk_wait    (VkSubmitInfo2& submit_info, u64 wait_value);
  void cuda_signal(cudaStream_t stream, u64 signal_value);
  void cuda_wait  (cudaStream_t stream, u64 wait_value);
};

// Per-frame sync state
struct InteropSync {
  SharedTimeline vk_to_cuda;   // Vulkan done → CUDA can read
  SharedTimeline cuda_to_vk;   // CUDA done → Vulkan can read
  u64            frame_index = 0;

  static InteropSync create(VkDevice vk_device);
  void               destroy(VkDevice vk_device);
  void               advance_frame();
};

### Interop Pattern — Frame Flow

```
Frame N:

[Vulkan] Rasterise coarse geometry → depth buffer, G-buffer
         signal(vk_to_cuda, frame_N)
         ↓
[CUDA]   wait(vk_to_cuda, frame_N)
         Ray march SDF scene from depth buffer hits
         Evaluate exact lighting, shadows, reflections
         Write to output texture (shared Vulkan/CUDA)
         signal(cuda_to_vk, frame_N)
         ↓
[Vulkan] wait(cuda_to_vk, frame_N)
         Composite CUDA output with Vulkan raster output
         Post-process, UI
         Present
```

## 2.7 — SDF Ray Marcher (CUDA)

The first CUDA render pass. Takes the depth buffer from Vulkan rasterisation and ray marches the SDF scene from each pixel's world-space position.

### Ray Marcher Kernel

```cpp
// compute/kernels/ray_march.cuh
#pragma once
#include "../../core/math/sdf.hpp"
#include "../../core/math/vec.hpp"
#include "../../core/types.hpp"
#include <cuda_runtime.h>
#include <surface_functions.h>

struct RayMarchParams {
  cudaSurfaceObject_t output;       // RGBA16F output — shared with Vulkan
  cudaSurfaceObject_t depth_in;     // Vulkan depth buffer read-back
  u32                 width, height;

  // Camera
  vec3 cam_pos;
  vec3 cam_forward, cam_right, cam_up;
  f32  fov_rad;

  // March settings — driven by GRIA alpha
  u32  max_steps;      // typical: 128
  f32  min_dist;       // hit threshold: 0.0001f
  f32  max_dist;       // far plane: 1000.f
  f32  alpha;          // GRIA α: 0 = full quality, 1 = fast approximate

  // Scene — SDF primitives (GPU-resident)
  struct SDFObject {
    vec3  position;
    vec3  params;        // r, or half_extents, etc.
    u32   type;          // 0=sphere 1=box 2=capsule 3=torus
    f32   alpha;         // per-object GRIA alpha for smooth blending
    u32   material_id;
  };
  const SDFObject* objects;
  u32              object_count;
};

__device__ f32 scene_sdf(const RayMarchParams& p, vec3 pos);
__device__ vec3 scene_normal(const RayMarchParams& p, vec3 hit);
__device__ f32 scene_shadow(const RayMarchParams& p, vec3 ro, vec3 rd, f32 tmin, f32 tmax);

__global__ void ray_march_kernel(RayMarchParams params);

// Host-side launcher
void launch_ray_march(const RayMarchParams& params, cudaStream_t stream);
```

### SDF Scene Evaluation

```cpp
__device__ f32 scene_sdf(const RayMarchParams& p, vec3 pos) {
  f32 d = p.max_dist;

  for (u32 i = 0; i < p.object_count; i++) {
    const auto& obj = p.objects[i];
    vec3 local = {pos.x - obj.position.x,
                  pos.y - obj.position.y,
                  pos.z - obj.position.z};
    f32 obj_d;
    switch (obj.type) {
      case 0: obj_d = SDF::sphere(local, obj.params.x); break;
      case 1: obj_d = SDF::box(local, obj.params);      break;
      case 2: obj_d = SDF::capsule(local, {0,0,0}, {0, obj.params.y, 0}, obj.params.x); break;
      case 3: obj_d = SDF::torus(local, obj.params.x, obj.params.y); break;
      default: obj_d = p.max_dist; break;
    }
    d = SDF::op_smooth_union(d, obj_d, obj.alpha);
  }
  return d;
}

__global__ void ray_march_kernel(RayMarchParams p) {
  u32 px = blockIdx.x * blockDim.x + threadIdx.x;
  u32 py = blockIdx.y * blockDim.y + threadIdx.y;
  if (px >= p.width || py >= p.height) return;

  // Construct ray from camera
  f32 u_ndc = (2.f * px / p.width  - 1.f) * tanf(p.fov_rad * 0.5f) * (f32)p.width / p.height;
  f32 v_ndc = (1.f - 2.f * py / p.height) * tanf(p.fov_rad * 0.5f);
  vec3 rd   = (p.cam_right * u_ndc + p.cam_up * v_ndc + p.cam_forward).normalized();
  vec3 ro   = p.cam_pos;

  // Sphere-tracing loop
  f32 t = 0.f;
  u32 steps = 0;
  bool hit = false;
  while (steps < p.max_steps && t < p.max_dist) {
    f32 d = scene_sdf(p, {ro.x + rd.x*t, ro.y + rd.y*t, ro.z + rd.z*t});
    if (d < p.min_dist) { hit = true; break; }
    t += d;
    steps++;
  }

  // Shade hit point
  float4 color = {0.05f, 0.05f, 0.08f, 1.f};  // background
  if (hit) {
    vec3 hit_pos = {ro.x + rd.x*t, ro.y + rd.y*t, ro.z + rd.z*t};
    vec3 n       = scene_normal(p, hit_pos);

    // Simple diffuse + ambient — full PBR added in Phase 5
    vec3 light_dir = vec3{0.5f, 1.f, 0.3f}.normalized();
    f32  diff      = fmaxf(0.f, n.dot(light_dir));
    f32  shadow    = scene_shadow(p, hit_pos, light_dir, 0.01f, 10.f);
    f32  ao        = 1.f - (f32)steps / p.max_steps * 0.4f;  // approx AO from step count

    color = {diff * shadow * ao, diff * shadow * ao * 0.9f, diff * shadow * ao * 0.8f, 1.f};
  }

  surf2Dwrite(color, p.output, px * sizeof(float4), py);
}
```

## 2.8 — Rasterisation Pipeline

The Vulkan raster side — renders proxy geometry for primary visibility, populates the depth buffer and G-buffer for CUDA to consume.

### G-Buffer Layout

```
Attachment 0: RGBA16F — world-space normal (xyz) + metallic (w)
Attachment 1: RGBA8   — base color (rgb) + roughness (a)
Attachment 2: R32F    — depth (for CUDA ray march seed)
Attachment 3: RG16F   — velocity (for TAA in Phase 5)
```

### Geometry Pass Shader (GLSL)

```glsl
// shaders/geometry_pass.frag.glsl
#version 460
#extension GL_EXT_nonuniform_qualifier : enable

layout(location = 0) in  vec3 frag_normal;
layout(location = 1) in  vec2 frag_uv;
layout(location = 2) in  vec3 frag_world_pos;

layout(location = 0) out vec4 g_normal_metallic;
layout(location = 1) out vec4 g_albedo_roughness;

// Bindless material data
layout(push_constant) uniform PC {
  uint material_id;
  mat4 model;
  mat4 mvp;
} pc;

layout(set = 0, binding = 1) uniform texture2D textures[];
layout(set = 0, binding = 2) uniform sampler   samplers[];

struct Material {
  vec4  base_color;
  float metallic;
  float roughness;
  uint  albedo_tex;
  uint  normal_tex;
};
layout(set = 0, binding = 3) readonly buffer Materials { Material data[]; } materials;

void main() {
  Material mat = materials.data[pc.material_id];
  vec4 albedo  = mat.base_color;
  if (mat.albedo_tex != ~0u) {
    albedo *= texture(sampler2D(textures[mat.albedo_tex], samplers[0]), frag_uv);
  }
  g_normal_metallic   = vec4(normalize(frag_normal) * 0.5 + 0.5, mat.metallic);
  g_albedo_roughness  = vec4(albedo.rgb, mat.roughness);
}
```

## 2.9 — Composite Pass

Merges the CUDA ray-marched output with the Vulkan raster output into the final frame.

```glsl
// shaders/composite.frag.glsl
#version 460
#extension GL_EXT_nonuniform_qualifier : enable

layout(location = 0) in  vec2 uv;
layout(location = 0) out vec4 out_color;

layout(push_constant) uniform PC {
  uint raster_texture;    // bindless index — Vulkan G-buffer composite
  uint cuda_texture;      // bindless index — CUDA ray-marched output
  uint depth_texture;
  float blend;            // GRIA alpha — 0=all CUDA, 1=all raster
} pc;

layout(set = 0, binding = 1) uniform texture2D textures[];
layout(set = 0, binding = 2) uniform sampler   samplers[];

void main() {
  vec4 raster = texture(sampler2D(textures[pc.raster_texture], samplers[0]), uv);
  vec4 cuda   = texture(sampler2D(textures[pc.cuda_texture],   samplers[0]), uv);
  // GRIA alpha blends raster ↔ CUDA output — tunable at runtime
  out_color   = mix(cuda, raster, pc.blend);
}
```

  JobScheduler::init();

  Renderer renderer;
  renderer.init({.window_handle = wnd.native, .width=1920, .height=1080});

  f64 last_time = Platform::get_time_seconds();

  while (Platform::poll_events()) {
    f64 now = Platform::get_time_seconds();
    f32 dt  = (f32)(now - last_time);

By the end of Phase 3 the engine renders a real scene, not just hand-authored test geometry.
